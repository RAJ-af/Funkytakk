-- 2026-10-30 fixes found while testing the app:
--  (1) 1:1 reactions never reached the other person: dm_messages.message_type CHECK did not allow 'reaction'
--  (2) voice notes / images could not be downloaded: there were NO storage policies for dm-media,
--      group-media and covers, so every upload was denied
--  (3) room-ended screen: the end screen could race with the participant rows being deleted at the same
--      moment; participants now stay ~1 minute after the end and the summary row is also published to Realtime
--  (4) discoverable_profiles was readable by anonymous (not signed in) callers

-- ---------------------------------------------------------------------------
-- (1) DM reactions
-- ---------------------------------------------------------------------------
alter table public.dm_messages drop constraint if exists dm_messages_message_type_check;
alter table public.dm_messages
  add constraint dm_messages_message_type_check
  check (message_type in ('text', 'image', 'voice', 'reaction'));

-- ---------------------------------------------------------------------------
-- (2) Storage policies (path contract: first folder = conversation_id / group_id / user_id)
--     dm-media:    <conversation_id>/<random>.bin
--     group-media: <group_id>/<random>.bin
--     covers:      <user_id>/<random>.<ext>
-- ---------------------------------------------------------------------------
drop policy if exists "dm participants can upload dm media" on storage.objects;
drop policy if exists "dm participants can read dm media" on storage.objects;
drop policy if exists "dm participants can delete dm media" on storage.objects;

create policy "dm participants can upload dm media" on storage.objects
  for insert to authenticated
  with check (
    bucket_id = 'dm-media'
    and exists (
      select 1 from public.dm_conversations c
      where c.id::text = (storage.foldername(name))[1]
        and (select auth.uid()) in (c.user_a, c.user_b)
    )
  );

create policy "dm participants can read dm media" on storage.objects
  for select to authenticated
  using (
    bucket_id = 'dm-media'
    and exists (
      select 1 from public.dm_conversations c
      where c.id::text = (storage.foldername(name))[1]
        and (select auth.uid()) in (c.user_a, c.user_b)
    )
  );

create policy "dm participants can delete dm media" on storage.objects
  for delete to authenticated
  using (
    bucket_id = 'dm-media'
    and exists (
      select 1 from public.dm_conversations c
      where c.id::text = (storage.foldername(name))[1]
        and (select auth.uid()) in (c.user_a, c.user_b)
    )
  );

drop policy if exists "group members can upload group media" on storage.objects;
drop policy if exists "group members can read group media" on storage.objects;
drop policy if exists "uploader can delete own group media" on storage.objects;

create policy "group members can upload group media" on storage.objects
  for insert to authenticated
  with check (
    bucket_id = 'group-media'
    and exists (
      select 1 from public.group_members m
      where m.group_id::text = (storage.foldername(name))[1]
        and m.user_id = (select auth.uid())
        and m.status = 'joined'
    )
  );

create policy "group members can read group media" on storage.objects
  for select to authenticated
  using (
    bucket_id = 'group-media'
    and exists (
      select 1 from public.group_members m
      where m.group_id::text = (storage.foldername(name))[1]
        and m.user_id = (select auth.uid())
        and m.status = 'joined'
    )
  );

-- group media is one shared encrypted file for many recipients: only the uploader may delete it;
-- everything else is removed by the 4-day purge jobs.
create policy "uploader can delete own group media" on storage.objects
  for delete to authenticated
  using (bucket_id = 'group-media' and owner = (select auth.uid()));

drop policy if exists "owner can upload own cover" on storage.objects;
drop policy if exists "owner can update own cover" on storage.objects;
drop policy if exists "owner can delete own cover" on storage.objects;
drop policy if exists "owner can read own cover" on storage.objects;

create policy "owner can upload own cover" on storage.objects
  for insert to authenticated
  with check (bucket_id = 'covers' and (storage.foldername(name))[1] = (select auth.uid())::text);
create policy "owner can read own cover" on storage.objects
  for select to authenticated
  using (bucket_id = 'covers' and (storage.foldername(name))[1] = (select auth.uid())::text);
create policy "owner can update own cover" on storage.objects
  for update to authenticated
  using (bucket_id = 'covers' and (storage.foldername(name))[1] = (select auth.uid())::text)
  with check (bucket_id = 'covers' and (storage.foldername(name))[1] = (select auth.uid())::text);
create policy "owner can delete own cover" on storage.objects
  for delete to authenticated
  using (bucket_id = 'covers' and (storage.foldername(name))[1] = (select auth.uid())::text);

-- ---------------------------------------------------------------------------
-- (3) Room-ended flow: summary in Realtime, participants kept ~1 minute after the end
-- ---------------------------------------------------------------------------
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'room_end_summaries'
  ) then
    alter publication supabase_realtime add table public.room_end_summaries;
  end if;
end $$;

create or replace function public.end_voice_room(p_room_id uuid)
returns jsonb
language plpgsql
security definer
set search_path to 'pg_catalog', 'public'
as $$
declare
  caller_id uuid := auth.uid();
  v_host_id uuid;
  caller_role text;
  v_reason text;
begin
  select host_id into v_host_id from public.voice_rooms where id = p_room_id;
  if not found then
    return jsonb_build_object('success', false, 'error', 'Room not found');
  end if;

  select role into caller_role from public.user_roles where user_id = caller_id::text;

  if caller_id != v_host_id and coalesce(caller_role, '') not in ('moderator', 'admin', 'superadmin') then
    raise exception 'Unauthorized: Only the host or moderator can end the room.';
  end if;

  v_reason := case when caller_id = v_host_id then 'host_ended' else 'moderation' end;

  update public.voice_rooms
     set is_live = false, status = 'ended', ended_at = now(),
         host_left_at = null, ended_reason = v_reason
   where id = p_room_id;

  perform private.finalize_room(p_room_id, v_reason);

  -- Do NOT delete participants here (clients would see "my row vanished" before "room ended").
  -- Mark them as left (this also books the RTC minutes); the cron job below deletes the rows.
  update public.room_participants set left_at = now()
   where room_id = p_room_id and left_at is null;
  delete from public.speaker_requests where room_id = p_room_id and status = 'pending';

  return jsonb_build_object('success', true, 'room_id', p_room_id, 'status', 'ended', 'ended_reason', v_reason);
end;
$$;

create or replace function public.check_and_cleanup_ended_rooms()
returns void
language plpgsql
security definer
set search_path to 'pg_catalog', 'public'
as $$
declare
  ids uuid[];
  rid uuid;
begin
  with u as (
    update public.voice_rooms
       set is_live = false, status = 'ended', ended_at = now(), ended_reason = 'host_timeout'
     where is_live = true
       and status = 'active'
       and host_left_at is not null
       and host_left_at < (now() - interval '3 minutes')
    returning id
  )
  select array_agg(id) into ids from u;

  if ids is not null then
    foreach rid in array ids loop
      perform private.finalize_room(rid, 'host_timeout');
      update public.room_participants set left_at = now()
       where room_id = rid and left_at is null;
    end loop;
  end if;

  -- rows of rooms that ended more than a minute ago can go now
  delete from public.room_participants
   where room_id in (
     select id from public.voice_rooms
      where (is_live = false or status = 'ended')
        and coalesce(ended_at, now() - interval '1 hour') < now() - interval '1 minute'
   );
end;
$$;
