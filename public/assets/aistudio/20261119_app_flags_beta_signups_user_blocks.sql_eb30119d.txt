-- ===== 1. app_flags: server switches the app reads at start (no new APK needed to flip) =====
create table if not exists public.app_flags (
  key         text primary key,
  value       jsonb not null,
  description text,
  updated_at  timestamptz not null default now()
);
alter table public.app_flags enable row level security;
drop policy if exists app_flags_read on public.app_flags;
create policy app_flags_read on public.app_flags for select to anon, authenticated using (true);
revoke all on public.app_flags from anon, authenticated;
grant select on public.app_flags to anon, authenticated;

insert into public.app_flags (key, value, description) values
  ('coin_store_enabled', 'false'::jsonb, 'Show the coin store (buying coins). Keep false until Google Play Billing is live.'),
  ('vip_enabled',        'false'::jsonb, 'Show VIP tags and the VIP banner upload tile.'),
  ('min_supported_version_code', '0'::jsonb, 'Installed versionCode below this number should see an update screen.')
on conflict (key) do nothing;

create or replace function public.get_app_flags()
returns jsonb
language sql
stable
security invoker
set search_path = pg_catalog, public
as $$
  select coalesce(jsonb_object_agg(key, value), '{}'::jsonb) from public.app_flags;
$$;
grant execute on function public.get_app_flags() to anon, authenticated;

-- ===== 2. beta signups (written only by the beta-signup Edge Function with the service role) =====
create table if not exists public.beta_signups (
  id         bigint generated always as identity primary key,
  email      text not null check (length(email) <= 254 and email = lower(email)),
  learning   text,
  android    boolean not null default true,
  ip_hash    text,
  source     text not null default 'landing',
  created_at timestamptz not null default now()
);
create unique index if not exists beta_signups_email_uniq on public.beta_signups (email);  -- emails are stored lower-case
alter table public.beta_signups enable row level security;
revoke all on public.beta_signups from anon, authenticated;

create table if not exists public.beta_signup_attempts (
  id         bigint generated always as identity primary key,
  ip_hash    text not null,
  created_at timestamptz not null default now()
);
create index if not exists beta_signup_attempts_ip_time on public.beta_signup_attempts (ip_hash, created_at desc);
alter table public.beta_signup_attempts enable row level security;
revoke all on public.beta_signup_attempts from anon, authenticated;

-- ===== 3. user blocking =====
create table if not exists public.user_blocks (
  blocker_id uuid not null,
  blocked_id uuid not null,
  created_at timestamptz not null default now(),
  primary key (blocker_id, blocked_id),
  check (blocker_id <> blocked_id)
);
create index if not exists user_blocks_blocked_idx on public.user_blocks (blocked_id);
alter table public.user_blocks enable row level security;
revoke all on public.user_blocks from anon, authenticated;   -- only the RPCs below touch it

create or replace function private.is_blocked_between(a uuid, b uuid)
returns boolean
language sql
stable
security definer
set search_path = pg_catalog, public
as $$
  select exists (
    select 1 from public.user_blocks
    where (blocker_id = a and blocked_id = b) or (blocker_id = b and blocked_id = a)
  );
$$;

-- true when EITHER side blocked the other (does not reveal who blocked whom)
create or replace function public.is_blocked_with(p_user_id uuid)
returns boolean
language plpgsql
stable
security definer
set search_path = pg_catalog, public
as $$
begin
  if auth.uid() is null then raise exception 'not_authenticated' using errcode = '28000'; end if;
  return private.is_blocked_between(auth.uid(), p_user_id);
end;
$$;

create or replace function public.block_user(p_user_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = pg_catalog, public
as $$
declare v_uid uuid := auth.uid();
begin
  if v_uid is null then raise exception 'not_authenticated' using errcode = '28000'; end if;
  if p_user_id is null or p_user_id = v_uid then raise exception 'cannot_block_yourself' using errcode = '22023'; end if;
  if not exists (select 1 from public.profiles where auth_user_id = p_user_id) then
    raise exception 'user_not_found' using errcode = '22023';
  end if;
  insert into public.user_blocks (blocker_id, blocked_id) values (v_uid, p_user_id) on conflict do nothing;
  -- blocking also removes follows in both directions
  delete from public.followers
   where (follower_id = v_uid and following_id = p_user_id)
      or (follower_id = p_user_id and following_id = v_uid);
  return jsonb_build_object('ok', true);
end;
$$;

create or replace function public.unblock_user(p_user_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = pg_catalog, public
as $$
begin
  if auth.uid() is null then raise exception 'not_authenticated' using errcode = '28000'; end if;
  delete from public.user_blocks where blocker_id = auth.uid() and blocked_id = p_user_id;
  return jsonb_build_object('ok', true);
end;
$$;

create or replace function public.get_my_blocked_users()
returns table (user_id uuid, username text, profile_name text, avatar_url text, blocked_at timestamptz)
language plpgsql
stable
security definer
set search_path = pg_catalog, public
as $$
begin
  if auth.uid() is null then raise exception 'not_authenticated' using errcode = '28000'; end if;
  return query
    select b.blocked_id, p.username, p.profile_name, p.avatar_url, b.created_at
    from public.user_blocks b
    left join public.profiles p on p.auth_user_id = b.blocked_id
    where b.blocker_id = auth.uid()
    order by b.created_at desc;
end;
$$;

revoke all on function public.is_blocked_with(uuid), public.block_user(uuid), public.unblock_user(uuid), public.get_my_blocked_users() from public, anon;
grant execute on function public.is_blocked_with(uuid), public.block_user(uuid), public.unblock_user(uuid), public.get_my_blocked_users() to authenticated;

-- ===== 4. enforce blocks in the three places that reach another user directly =====
create or replace function public.get_or_create_dm_conversation(p_other_user_id uuid)
 returns uuid
 language plpgsql
 security definer
 set search_path to 'pg_catalog', 'public'
as $function$
declare
  v_uid uuid := auth.uid();
  v_a uuid;
  v_b uuid;
  v_id uuid;
begin
  if v_uid is null then
    raise exception 'not_authenticated' using errcode = '28000';
  end if;
  if p_other_user_id = v_uid then
    raise exception 'cannot_dm_yourself' using errcode = '22023';
  end if;
  if private.is_blocked_between(v_uid, p_other_user_id) then
    raise exception 'user_blocked' using errcode = '42501';
  end if;

  if v_uid < p_other_user_id then
    v_a := v_uid; v_b := p_other_user_id;
  else
    v_a := p_other_user_id; v_b := v_uid;
  end if;

  insert into public.dm_conversations (user_a, user_b)
  values (v_a, v_b)
  on conflict (user_a, user_b) do nothing;

  select id into v_id from public.dm_conversations where user_a = v_a and user_b = v_b;
  return v_id;
end;
$function$;

create or replace function public.send_gift(p_recipient_id uuid, p_gift_id bigint, p_message text default null::text)
 returns jsonb
 language plpgsql
 security definer
 set search_path to 'pg_catalog', 'public'
as $function$
declare
  v_uid uuid := auth.uid();
  v_price integer;
  v_balance integer;
begin
  if v_uid is null then
    raise exception 'not_authenticated' using errcode = '28000';
  end if;
  if p_recipient_id = v_uid then
    raise exception 'cannot_gift_yourself' using errcode = '22023';
  end if;
  if private.is_blocked_between(v_uid, p_recipient_id) then
    raise exception 'user_blocked' using errcode = '42501';
  end if;

  select price_coins into v_price from public.gift_catalog where id = p_gift_id and is_active = true;
  if v_price is null then
    raise exception 'gift_not_found' using errcode = '22023';
  end if;

  insert into public.user_wallets (user_id) values (v_uid) on conflict (user_id) do nothing;

  update public.user_wallets
  set coins_balance = coins_balance - v_price, updated_at = now()
  where user_id = v_uid and coins_balance >= v_price
  returning coins_balance into v_balance;

  if v_balance is null then
    raise exception 'insufficient_coins' using errcode = 'P0001';
  end if;

  insert into public.gift_transactions (sender_id, recipient_id, gift_id, coins_spent, message)
  values (v_uid, p_recipient_id, p_gift_id, v_price, p_message);

  insert into public.notifications (user_id, type, title, message)
  values (p_recipient_id, 'gift_received', 'You received a gift!', coalesce(p_message, 'Someone sent you a gift.'));

  return jsonb_build_object('ok', true, 'remaining_balance', v_balance);
end;
$function$;

create or replace function public.send_room_invite(p_room_id uuid, p_user_id uuid)
 returns jsonb
 language plpgsql
 security definer
 set search_path to 'pg_catalog', 'public'
as $function$
declare
  caller uuid := auth.uid();
  v_title text;
  v_live boolean;
  v_name text;
  v_existing record;
  v_id uuid;
begin
  if caller is null then
    raise exception 'not_authenticated' using errcode = '28000';
  end if;

  select title, (is_live and status = 'active') into v_title, v_live
    from public.voice_rooms where id = p_room_id;
  if not found then
    return jsonb_build_object('success', false, 'error', 'room_not_found');
  end if;
  if not coalesce(v_live, false) then
    return jsonb_build_object('success', false, 'error', 'room_not_live');
  end if;

  if not exists (select 1 from public.room_participants where room_id = p_room_id and user_id = caller) then
    return jsonb_build_object('success', false, 'error', 'not_in_room');
  end if;
  if p_user_id = caller then
    return jsonb_build_object('success', false, 'error', 'cannot_invite_self');
  end if;
  if private.is_blocked_between(caller, p_user_id) then
    return jsonb_build_object('success', false, 'error', 'user_blocked');
  end if;
  if not exists (select 1 from public.followers where follower_id = caller and following_id = p_user_id) then
    return jsonb_build_object('success', false, 'error', 'not_following');
  end if;
  if exists (select 1 from public.room_participants where room_id = p_room_id and user_id = p_user_id) then
    return jsonb_build_object('success', false, 'error', 'already_in_room');
  end if;

  if (select count(*) from public.room_invites
       where inviter_id = caller and created_at > now() - interval '1 hour') >= 30 then
    return jsonb_build_object('success', false, 'error', 'rate_limited');
  end if;

  select * into v_existing from public.room_invites
   where room_id = p_room_id and invitee_id = p_user_id;
  if found and v_existing.status = 'pending' and v_existing.created_at > now() - interval '60 seconds' then
    return jsonb_build_object('success', false, 'error', 'already_invited', 'invite_id', v_existing.id);
  end if;

  insert into public.room_invites (room_id, inviter_id, invitee_id, status, created_at)
  values (p_room_id, caller, p_user_id, 'pending', now())
  on conflict (room_id, invitee_id)
  do update set inviter_id = excluded.inviter_id, status = 'pending', created_at = now()
  returning id into v_id;

  select username into v_name from public.profiles where auth_user_id = caller;

  insert into public.notifications (user_id, type, title, message, room_id)
  values (p_user_id, 'room_invite',
          coalesce(v_name, 'Someone') || ' invited you to a voice room',
          coalesce(v_title, 'Voice room'), p_room_id);

  return jsonb_build_object('success', true, 'invite_id', v_id, 'room_id', p_room_id, 'user_id', p_user_id);
end;
$function$;
