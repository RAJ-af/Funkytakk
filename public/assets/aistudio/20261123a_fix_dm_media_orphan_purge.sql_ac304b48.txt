-- Applied live on 2026-11-23.
-- Problem: cron job purge-orphaned-dm-media-daily failed every night ("Direct deletion from storage tables is not allowed. Use the Storage API"),
-- so unreferenced encrypted chat blobs in bucket dm-media were never removed (10 blobs, oldest from 2026-11-05).
-- Fix: SQL only LISTS orphans, the Edge Function process-media-cleanup (job "dm_orphans") removes them through the Storage API.
create or replace function public.list_orphaned_dm_media(p_older_than_days integer default 4, p_limit integer default 500)
returns table (name text)
language sql stable security definer
set search_path = pg_catalog, public, storage
as $f$
  select o.name
  from storage.objects o
  where o.bucket_id = 'dm-media'
    and o.created_at < now() - make_interval(days => greatest(p_older_than_days, 1))
    and not exists (select 1 from public.dm_messages dm where dm.storage_path = o.name)
  order by o.created_at
  limit least(greatest(p_limit, 1), 1000)
$f$;
revoke all on function public.list_orphaned_dm_media(integer, integer) from public, anon, authenticated;
grant execute on function public.list_orphaned_dm_media(integer, integer) to service_role;

select cron.schedule('purge-orphaned-dm-media-daily', '15 3 * * *', $$select private.call_edge('process-media-cleanup', '{"job":"dm_orphans"}'::jsonb)$$);
